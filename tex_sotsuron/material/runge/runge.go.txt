package main

import (
	"fmt"
	"image/color"
	"math"

	"gonum.org/v1/plot"
	"gonum.org/v1/plot/plotter"
	"gonum.org/v1/plot/vg"
)

// TargetFunc: 近似対象の関数 (ルンゲ関数)
// f(x) = 1 / (1 + 25x^2)
func TargetFunc(x float64) float64 {
	return 1.0 / (1.0 + 25.0*x*x)
}

func main() {
	fmt.Println("ルンゲ現象シミュレーション（誤差グラフ生成モード）を開始します...")

	rangeStart := -1.0
	rangeEnd := -1.0 * rangeStart // 1.0

	// 評価・描画に使用するグリッド数
	gridSteps := 1000

	maxN := 25

	for n := 1; n <= maxN; n++ {
		// 1. 近似多項式のプロット (元の名前のまま)
		rungeFile := fmt.Sprintf("runge_n%02d.png", n)
		if err := plotRunge(n, rangeStart, rangeEnd, gridSteps, rungeFile); err != nil {
			fmt.Printf("Runge Plot Error: %v\n", err)
		}

		// 2. 誤差関数のプロット (指定された名前に変更)
		errorFile := fmt.Sprintf("runge_err_n%02d.png", n)
		if err := plotError(n, rangeStart, rangeEnd, gridSteps, errorFile); err != nil {
			fmt.Printf("Error Plot Error: %v\n", err)
		} else {
			fmt.Printf("N=%02d: Saved %s and %s\n", n, rungeFile, errorFile)
		}
	}
}

// plotRunge: 真の関数と補間多項式をプロット
func plotRunge(n int, a, b float64, steps int, filename string) error {
	p := plot.New()

	p.Title.Text = fmt.Sprintf("Runge Phenomenon (N=%d)", n)
	p.X.Label.Text = "x"
	p.Y.Label.Text = "y"
	p.Y.Min, p.Y.Max = -1.0, 1.5 // 近似の様子が見やすい固定範囲

	// 1. 真の関数の描画
	trueLineData := make(plotter.XYs, steps+1)
	for i := 0; i <= steps; i++ {
		x := a + (b-a)*float64(i)/float64(steps)
		trueLineData[i] = plotter.XY{X: x, Y: TargetFunc(x)}
	}
	lTrue, _ := plotter.NewLine(trueLineData)
	lTrue.Color = color.RGBA{R: 0, G: 0, B: 0, A: 255} // 黒
	p.Add(lTrue)
	p.Legend.Add("Original f(x)", lTrue)

	// 2. 補間多項式の描画
	uniX, uniY := generateEquidistantNodes(n, a, b)
	uniLineData := interpolateLine(uniX, uniY, a, b, steps)

	lUni, _ := plotter.NewLine(uniLineData)
	lUni.Color = color.RGBA{R: 255, G: 0, B: 0, A: 255} // 赤
	lUni.Dashes = []vg.Length{vg.Points(2), vg.Points(2)}
	p.Add(lUni)
	p.Legend.Add(fmt.Sprintf("Interpolation P%d(x)", n), lUni)

	return p.Save(6*vg.Inch, 4*vg.Inch, filename)
}

// plotError: 誤差関数 e(x) = Pn(x) - f(x) をプロット
func plotError(n int, a, b float64, steps int, filename string) error {
	p := plot.New()

	p.Title.Text = fmt.Sprintf("Interpolation Error (N=%d)", n)
	p.X.Label.Text = "x"
	p.Y.Label.Text = "e(x) = Pn(x) - f(x)"

	// ノード生成と誤差計算
	uniX, uniY := generateEquidistantNodes(n, a, b)
	errorData := make(plotter.XYs, steps+1)

	maxAbsErr := 0.0
	for i := 0; i <= steps; i++ {
		x := a + (b-a)*float64(i)/float64(steps)
		yApprox := lagrangeVal(x, uniX, uniY)
		yTrue := TargetFunc(x)
		errVal := yApprox - yTrue
		errorData[i] = plotter.XY{X: x, Y: errVal}

		if math.Abs(errVal) > maxAbsErr {
			maxAbsErr = math.Abs(errVal)
		}
	}

	// 誤差グラフの線
	lError, _ := plotter.NewLine(errorData)
	lError.Color = color.RGBA{R: 0, G: 100, B: 200, A: 255} // 青系
	p.Add(lError)

	// ゼロライン（基準線）の追加
	zeroLine, _ := plotter.NewLine(plotter.XYs{{X: a, Y: 0}, {X: b, Y: 0}})
	zeroLine.Color = color.RGBA{R: 150, G: 150, B: 150, A: 150}
	p.Add(zeroLine)

	// 最大絶対誤差を凡例に表示
	p.Legend.Add(fmt.Sprintf("Max Abs Error: %.2e", maxAbsErr), lError)

	return p.Save(6*vg.Inch, 4*vg.Inch, filename)
}

// generateEquidistantNodes: 等間隔ノード生成
func generateEquidistantNodes(n int, a, b float64) ([]float64, []float64) {
	xs := make([]float64, n+1)
	ys := make([]float64, n+1)
	for i := 0; i <= n; i++ {
		x := a + (b-a)*float64(i)/float64(n)
		xs[i] = x
		ys[i] = TargetFunc(x)
	}
	return xs, ys
}

// interpolateLine: ラグランジュ補間曲線用データ生成
func interpolateLine(nodesX, nodesY []float64, a, b float64, steps int) plotter.XYs {
	pts := make(plotter.XYs, steps+1)
	for i := 0; i <= steps; i++ {
		x := a + (b-a)*float64(i)/float64(steps)
		y := lagrangeVal(x, nodesX, nodesY)
		pts[i] = plotter.XY{X: x, Y: y}
	}
	return pts
}

// lagrangeVal: ラグランジュ補間多項式の値を計算
func lagrangeVal(x float64, xs, ys []float64) float64 {
	n := len(xs)
	sum := 0.0
	for i := 0; i < n; i++ {
		term := ys[i]
		for j := 0; j < n; j++ {
			if i != j {
				term *= (x - xs[j]) / (xs[i] - xs[j])
			}
		}
		sum += term
	}
	return sum
}
